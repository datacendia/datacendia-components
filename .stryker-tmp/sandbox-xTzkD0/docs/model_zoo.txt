/**
 * Helper to get the actual model string for Ollama
 */
export function getModelForAgent(agentId: string): string {
  const agent = AI_CONFIG.agents[agentId as keyof typeof AI_CONFIG.agents];
  if (!agent) return AI_CONFIG.models.flagship.id; // Default to smartest
  
  const modelType = agent.model as keyof typeof AI_CONFIG.models;
  return AI_CONFIG.models[modelType].id;
}
